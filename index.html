# 1. Unzip plant_tracker_local.zip and install FastAPI + Uvicorn

# 2. Start the local SQLite server (accessible on PC & iPhone over Wi-Fi)
python app.py
# Open http://localhost:8000 on PC or http://<PC-LAN-IP>:8000 on iPhone
